Deploy

One node runs four containers from published images: Caddy with HTTPS, Wrench, PostgreSQL and backup. Only Caddy publishes ports.

You need

  • A Linux host with Docker Compose, a DNS name pointing at it, ports 80 and 443 open.
  • A GitLab application: confidential, scopes openid profile email, callback https://<domain>/auth/gitlab/callback.
  • A GitLab token with read_api for account checks.
  • An OpenRouter key with a credit limit.
  • An S3 bucket for backups.

Files

/opt/wrench/
├── compose.yaml
├── .env                    # secrets
├── Caddyfile
├── initdb/10-backup-role.sh
├── config/config.yaml      # policy, see Configure
└── data/                   # all state: database, certificates, local backups
    ├── postgres/
    ├── caddy/
    └── backup/

Everything Wrench keeps is in data/ on the host, so it is plain files you can see, move and include in host-level backups. Copy the restic password and S3 keys from .env somewhere off the node: a restore needs them.

compose.yaml

Each container gets only its own secrets: the OpenRouter key reaches Wrench only, the S3 keys reach backup only.

# Wrench on one node: Caddy (HTTPS), Wrench, PostgreSQL and backup.
# Secrets come from .env next to this file, policy from config/, state lives in data/.
name: wrench

services:
  caddy:
    image: caddy:2.10.2-alpine@sha256:4c6e91c6ed0e2fa03efd5b44747b625fec79bc9cd06ac5235a779726618e530d
    restart: unless-stopped
    environment:
      WRENCH_DOMAIN: ${WRENCH_DOMAIN:?}
    ports:
      - "80:80"
      - "443:443"
      - "443:443/udp"
    volumes:
      - ./Caddyfile:/etc/caddy/Caddyfile:ro
      - ./data/caddy/data:/data
      - ./data/caddy/config:/config
    networks: [edge]
    depends_on:
      wrench:
        condition: service_healthy

  wrench:
    image: ghcr.io/ordo-systems/wrench:${WRENCH_VERSION:?}
    restart: unless-stopped
    environment:
      WRENCH_LISTEN_ADDR: 0.0.0.0:8080
      WRENCH_PUBLIC_URL: https://${WRENCH_DOMAIN:?}
      WRENCH_WEB_SESSION_TTL: 12h
      WRENCH_CLIENT_KEYS_TTL: 168h
      WRENCH_POSTGRES_DSN: postgres://wrench:${POSTGRES_PASSWORD:?}@postgres:5432/wrench?sslmode=disable
      WRENCH_GITLAB_URL: ${WRENCH_GITLAB_URL:?}
      WRENCH_GITLAB_CLIENT_ID: ${WRENCH_GITLAB_CLIENT_ID:?}
      WRENCH_GITLAB_CLIENT_SECRET: ${WRENCH_GITLAB_CLIENT_SECRET:?}
      WRENCH_GITLAB_SYNC_TOKEN: ${WRENCH_GITLAB_SYNC_TOKEN:?}
      WRENCH_GITLAB_SYNC_INTERVAL: 5m
      WRENCH_OPENROUTER_KEY: ${WRENCH_OPENROUTER_KEY:?}
      WRENCH_UPSTREAM_TIMEOUT_CONNECT: 10s
      WRENCH_UPSTREAM_TIMEOUT_FIRST_TOKEN: 90s
      WRENCH_UPSTREAM_TIMEOUT_STREAM_IDLE: 90s
      WRENCH_UPSTREAM_TIMEOUT_TOTAL: 10m
      WRENCH_UPSTREAM_CONCURRENCY_PER_USER: 3
      WRENCH_UPSTREAM_CONCURRENCY_GLOBAL: 60
      WRENCH_SESSION_OFFLINE_AFTER: 60s
      WRENCH_ALERT_WEBHOOK_URL: ${WRENCH_ALERT_WEBHOOK_URL:-}
      WRENCH_BACKUP_STATUS_FILE: /var/lib/wrench/backup-status/backup.json
    volumes:
      - ./config:/etc/wrench:ro
      - ./data/backup/status:/var/lib/wrench/backup-status:ro
    networks: [edge, backend]
    depends_on:
      postgres:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "/usr/local/bin/wrench", "healthcheck", "http://127.0.0.1:8080/healthz"]
      interval: 10s
      timeout: 5s
      retries: 30
      start_period: 20s

  postgres:
    image: postgres:17-alpine@sha256:3540c0362c809df084ffac0ae69fe794515f9de92692f02178bcdce0ef0302ac
    restart: unless-stopped
    environment:
      POSTGRES_USER: wrench
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:?}
      POSTGRES_DB: wrench
      WRENCH_BACKUP_DB_PASSWORD: ${WRENCH_BACKUP_DB_PASSWORD:?}
    volumes:
      - ./data/postgres:/var/lib/postgresql/data
      - ./initdb:/docker-entrypoint-initdb.d:ro
    networks: [backend]
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U wrench -d wrench"]
      interval: 5s
      timeout: 3s
      retries: 30

  backup:
    image: ghcr.io/ordo-systems/wrench:${WRENCH_VERSION:?}
    command: ["run", "backup"]
    restart: unless-stopped
    environment:
      WRENCH_BACKUP_POSTGRES_DSN: postgres://wrench_backup:${WRENCH_BACKUP_DB_PASSWORD:?}@postgres:5432/wrench?sslmode=disable
      WRENCH_BACKUP_SCHEDULE: "0 2 * * *"
      WRENCH_BACKUP_RETENTION: 30d
      WRENCH_BACKUP_RESTIC_PASSWORD: ${WRENCH_BACKUP_RESTIC_PASSWORD:?}
      WRENCH_BACKUP_S3_ENDPOINT: ${WRENCH_BACKUP_S3_ENDPOINT:?}
      WRENCH_BACKUP_S3_BUCKET: ${WRENCH_BACKUP_S3_BUCKET:?}
      WRENCH_BACKUP_S3_PATH: wrench
      WRENCH_BACKUP_S3_REGION: ${WRENCH_BACKUP_S3_REGION:?}
      WRENCH_BACKUP_S3_ACCESS_KEY: ${WRENCH_BACKUP_S3_ACCESS_KEY:?}
      WRENCH_BACKUP_S3_SECRET_KEY: ${WRENCH_BACKUP_S3_SECRET_KEY:?}
      WRENCH_BACKUP_HEARTBEAT_URL: ${WRENCH_BACKUP_HEARTBEAT_URL:-}
    volumes:
      - ./config:/etc/wrench:ro
      - ./data/backup/restic:/var/lib/wrench-backup/restic
      - ./data/backup/status:/var/lib/wrench-backup/status
    networks: [backend, egress]
    depends_on:
      postgres:
        condition: service_healthy

# edge reaches the internet (GitLab, OpenRouter, webhook); backend is internal; egress lets backup reach S3.
networks:
  edge: {}
  backend:
    internal: true
  egress: {}

Use letters and digits for the two database passwords: they are embedded in connection URLs.

Caddyfile

# Public HTTPS entry point. Caddy obtains and renews the certificate for WRENCH_DOMAIN automatically;
# ports 80 and 443 must reach this host. Responses are streamed unbuffered for server-sent events,
# and nothing is compressed so streams are never held back.
{$WRENCH_DOMAIN} {
	reverse_proxy wrench:8080 {
		flush_interval -1
	}
}

initdb/10-backup-role.sh

Postgres runs it once, on an empty data/postgres: a read-only role for backups.

#!/bin/sh
# Runs once, when data/postgres is empty: a login role that can only read, used by the backup container.
set -eu
psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "$POSTGRES_DB" -v pw="$WRENCH_BACKUP_DB_PASSWORD" <<'SQL'
CREATE ROLE wrench_backup LOGIN PASSWORD :'pw';
GRANT pg_read_all_data TO wrench_backup;
SQL

config/config.yaml

Start

cd /opt/wrench
mkdir -p data/postgres data/caddy/data data/caddy/config data/backup/restic data/backup/status
sudo chown -R 65532:65532 data/backup         # the Wrench image runs as UID 65532
docker compose up -d
docker compose exec backup wrench backup     # first backup

Open https://<domain>/healthz, sign in to the dashboard, and point an external uptime check at /healthz: a stopped node cannot alert.

Images are ghcr.io/ordo-systems/wrench:<version>; edge follows master. Verify one with gh attestation verify oci://ghcr.io/ordo-systems/wrench:<version> --repo ordo-systems/wrench.
